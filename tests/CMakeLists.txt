# In a process of its own, with the final hash written down a second time: the golden file alone
# could be regenerated by mistake, this line would have to be changed by hand.
add_test(NAME headless_golden_defence COMMAND qtd_headless
    --map "${PROJECT_SOURCE_DIR}/content/training_ground.json"
    --replay "${QTD_GOLDEN_DIR}/training_ground_defence.replay.json")
set_tests_properties(headless_golden_defence PROPERTIES
    LABELS golden
    PASS_REGULAR_EXPRESSION "\"hash\":\"2aa462d18aa4fe67\".*\"matches\":true")
